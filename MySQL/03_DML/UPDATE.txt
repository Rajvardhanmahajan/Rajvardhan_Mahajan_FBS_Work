##Update employee salary
UPDATE employee
SET salary = 40000
WHERE emp_id = 101;


##Update multiple columns
UPDATE employee
SET salary = 45000,
    city = 'Mumbai'
WHERE emp_id = 101;


##Increase salary of all IT employees
UPDATE employee
SET salary = salary + 5000
WHERE dept_id = 1;


##Increase salary by 10%
UPDATE employee
SET salary = salary * 1.10
WHERE dept_id = 1;


##Update employees of multiple departments
UPDATE employee
SET salary = salary + 3000
WHERE dept_id IN (1, 3);


##Update using multiple conditions
UPDATE employee
SET salary = 50000
WHERE dept_id = 1
AND age > 23;


##Update Department Name
UPDATE department
SET dept_name = 'Human Resources'
WHERE dept_id = 2;


##Update dept_id of an employee
UPDATE employee
SET dept_id = 3
WHERE emp_id = 102;
