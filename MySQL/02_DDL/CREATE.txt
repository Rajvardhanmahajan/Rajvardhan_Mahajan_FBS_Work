-- Create Database
CREATE DATABASE fbs;

-- Use Database
USE fbs;

-- Create Table
CREATE TABLE employee (
    emp_id INT PRIMARY KEY,
    name VARCHAR(50),
    salary DECIMAL(10,2),
    age INT,
    gender CHAR(1),
    city VARCHAR(30),
    doj DATE
);

-- Create Table with Constraints
CREATE TABLE department (
    dept_id INT PRIMARY KEY,
    dept_name VARCHAR(25) NOT NULL UNIQUE,
    dept_loc VARCHAR(25)
);
